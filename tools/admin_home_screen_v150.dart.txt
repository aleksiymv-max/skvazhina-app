import 'package:flutter/material.dart';
import 'package:go_router/go_router.dart';
import '../../core/models/user.dart';
import '../../core/services/session_service.dart';
class AdminHomeScreen extends StatefulWidget{const AdminHomeScreen({super.key});@override State<AdminHomeScreen> createState()=>_AdminHomeScreenState();}
class _AdminHomeScreenState extends State<AdminHomeScreen>{final s=SessionService.instance;int tab=0;@override void initState(){super.initState();s.addListener(_r);}@override void dispose(){s.removeListener(_r);super.dispose();}void _r(){if(mounted)setState((){});}@override Widget build(BuildContext c)=>Scaffold(appBar:AppBar(title:const Text('Администрирование'),actions:[IconButton(onPressed:()async{await s.logout();if(c.mounted)c.go('/login');},icon:const Icon(Icons.logout))]),body:[_home(),_objects(),_users(),_requests(),_access()][tab],bottomNavigationBar:NavigationBar(selectedIndex:tab,onDestinationSelected:(v)=>setState(()=>tab=v),destinations:const [NavigationDestination(icon:Icon(Icons.dashboard_outlined),label:'Главная'),NavigationDestination(icon:Icon(Icons.domain_outlined),label:'Объекты'),NavigationDestination(icon:Icon(Icons.people_outline),label:'Пользователи'),NavigationDestination(icon:Icon(Icons.how_to_reg_outlined),label:'Заявки'),NavigationDestination(icon:Icon(Icons.admin_panel_settings_outlined),label:'Доступ') ]));
Widget _home()=>ListView(padding:const EdgeInsets.all(16),children:[const Text('Система «Промысел»',style:TextStyle(fontSize:24,fontWeight:FontWeight.w900)),const Text('Системное управление без производственных действий.'),const SizedBox(height:12),_tile('Месторождения',s.fields.length.toString(),Icons.domain),_tile('Пользователи',s.allWorkers.length.toString(),Icons.people),_tile('Ожидают согласования',s.allPending.length.toString(),Icons.pending_actions)]);
Widget _tile(String a,String b,IconData i)=>Card(child:ListTile(leading:Icon(i),title:Text(a),trailing:Text(b,style:const TextStyle(fontSize:22,fontWeight:FontWeight.w900))));
Widget _objects()=>ListView(padding:const EdgeInsets.all(16),children:[Row(children:[const Expanded(child:Text('Организации и месторождения',style:TextStyle(fontSize:22,fontWeight:FontWeight.w900))),FilledButton.icon(onPressed:_addField,icon:const Icon(Icons.add),label:const Text('Добавить'))]),...s.fields.map((f)=>Card(child:ListTile(leading:const Icon(Icons.domain),title:Text(f.name),subtitle:Text(s.chiefsForField(f.id).isEmpty?'Начальник цеха не назначен':'Начальник цеха назначен'))))]);
Widget _users()=>ListView(padding:const EdgeInsets.all(16),children:[const Text('Пользователи',style:TextStyle(fontSize:22,fontWeight:FontWeight.w900)),...s.allWorkers.map((u)=>Card(child:ListTile(title:Text(u.name),subtitle:Text(u.role.label+' · '+s.fieldName(s.userFields[u.id]??'')+(s.blockedUsers.contains(u.id)?' · ЗАБЛОКИРОВАН':'')),trailing:PopupMenuButton<String>(onSelected:(v){if(v=='edit')_edit(u);if(v=='block')s.adminSetBlocked(u.id,!s.blockedUsers.contains(u.id));},itemBuilder:(_)=>[const PopupMenuItem(value:'edit',child:Text('Роль и привязка')),PopupMenuItem(value:'block',child:Text(s.blockedUsers.contains(u.id)?'Восстановить':'Заблокировать'))]))))]);
Widget _requests()=>ListView(padding:const EdgeInsets.all(16),children:[const Text('Регистрации',style:TextStyle(fontSize:22,fontWeight:FontWeight.w900)),if(s.allPending.isEmpty)const Card(child:Padding(padding:EdgeInsets.all(18),child:Text('Заявок нет.'))),...s.allPending.map((p)=>Card(child:ListTile(title:Text(p.name),subtitle:Text(p.role.label+' · '+s.fieldName(p.fieldId)),trailing:FilledButton(onPressed:()=>s.approve(p.id),child:const Text('Согласовать')))))]);
Widget _access()=>ListView(padding:const EdgeInsets.all(16),children:const [Text('Роли и доступы',style:TextStyle(fontSize:22,fontWeight:FontWeight.w900)),Card(child:Padding(padding:EdgeInsets.all(16),child:Text('Администратор управляет пользователями, регистрациями и организационной структурой. Обход, ввод и корректировка замеров, производственные задания и закрытие проблем ему недоступны.'))),Card(child:ListTile(title:Text('Производственные роли'),subtitle:Text('Оператор ДНГ · Диспетчер · Мастер · Геолог · Ведущий геолог · Технолог · Ведущий технолог · Начальник цеха')))]);
Future<void> _addField()async{final x=TextEditingController();final ok=await showDialog<bool>(context:context,builder:(d)=>AlertDialog(title:const Text('Новое месторождение'),content:TextField(controller:x),actions:[TextButton(onPressed:()=>Navigator.pop(d,false),child:const Text('Отмена')),FilledButton(onPressed:()=>Navigator.pop(d,true),child:const Text('Создать'))]));if(ok==true)await s.adminAddField(x.text);x.dispose();}
Future<void> _edit(AppUser u)async{var role=u.role==UserRole.admin?UserRole.operator:u.role;var field=s.userFields[u.id]??s.fields.first.id;await showDialog(context:context,builder:(d)=>StatefulBuilder(builder:(d,setL)=>AlertDialog(title:Text(u.name),content:Column(mainAxisSize:MainAxisSize.min,children:[DropdownButtonFormField<UserRole>(initialValue:role,items:const [DropdownMenuItem(value:UserRole.operator,child:Text('Оператор ДНГ')),DropdownMenuItem(value:UserRole.dispatcher,child:Text('Диспетчер')),DropdownMenuItem(value:UserRole.engineer,child:Text('ИТР')),DropdownMenuItem(value:UserRole.shopChief,child:Text('Начальник цеха'))],onChanged:(v)=>setL(()=>role=v??role)),DropdownButtonFormField<String>(initialValue:field,items:s.fields.map((f)=>DropdownMenuItem(value:f.id,child:Text(f.name))).toList(),onChanged:(v)=>setL(()=>field=v??field))]),actions:[FilledButton(onPressed:()async{await s.adminChangeUser(u.id,role:role,fieldId:field,chiefId:s.chiefsForField(field).firstOrNull?.id);if(d.mounted)Navigator.pop(d);},child:const Text('Сохранить'))])));}
}
